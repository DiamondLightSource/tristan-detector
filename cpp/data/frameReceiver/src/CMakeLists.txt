set(CMAKE_INCLUDE_CURRENT_DIR on)
set(CMAKE_BUILD_TYPE Debug)

include_directories(${COMMON_DIR}/include ${FRAMERECEIVER_DIR}/include ${ODINDATA_INCLUDE_DIRS} ${Boost_INCLUDE_DIRS} ${LOG4CXX_INCLUDE_DIRS}/.. ${ZEROMQ_INCLUDE_DIRS})

# Add library for Tristan decoder
add_library(TristanFrameDecoder SHARED TristanFrameDecoder.cpp TristanFrameDecoderLib.cpp)

string(TIMESTAMP EXEC_TIME)
set(TRISTAN_EXEC_SCRIPT ${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/execute_tristan_receiver)
file(WRITE ${TRISTAN_EXEC_SCRIPT} "#!/bin/bash\n")
file(APPEND ${TRISTAN_EXEC_SCRIPT} "\n")
file(APPEND ${TRISTAN_EXEC_SCRIPT} "# This file is autogenerated - DO NOT EDIT\n")
file(APPEND ${TRISTAN_EXEC_SCRIPT} "# Generated on ${EXEC_TIME}\n")
file(APPEND ${TRISTAN_EXEC_SCRIPT} "\n")
file(APPEND ${TRISTAN_EXEC_SCRIPT} "# Executes the Tristan ODIN-DATA FrameReceiver\n")
file(APPEND ${TRISTAN_EXEC_SCRIPT} "\n")
file(APPEND ${TRISTAN_EXEC_SCRIPT} "${ODINDATA_ROOT_DIR}/bin/frameReceiver --path ${CMAKE_INSTALL_PREFIX}/lib -t TRISTAN \"$@\"")

install(
	FILES ${TRISTAN_EXEC_SCRIPT}
	PERMISSIONS OWNER_EXECUTE OWNER_READ
	DESTINATION bin
	)

install(TARGETS TristanFrameDecoder LIBRARY DESTINATION lib)
