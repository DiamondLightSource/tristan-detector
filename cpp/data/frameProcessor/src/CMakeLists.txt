string(TIMESTAMP EXEC_TIME)
set(TRISTAN_EXEC_SCRIPT ${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/execute_tristan_processor)
file(WRITE ${TRISTAN_EXEC_SCRIPT} "#!/bin/bash\n")
file(APPEND ${TRISTAN_EXEC_SCRIPT} "\n")
file(APPEND ${TRISTAN_EXEC_SCRIPT} "# This file is autogenerated - DO NOT EDIT\n")
file(APPEND ${TRISTAN_EXEC_SCRIPT} "# Generated on ${EXEC_TIME}\n")
file(APPEND ${TRISTAN_EXEC_SCRIPT} "\n")
file(APPEND ${TRISTAN_EXEC_SCRIPT} "# Executes the Tristan ODIN-DATA FrameProcessor\n")
file(APPEND ${TRISTAN_EXEC_SCRIPT} "\n")
file(APPEND ${TRISTAN_EXEC_SCRIPT} "cwd=$(pwd)\n")
file(APPEND ${TRISTAN_EXEC_SCRIPT} "cd ${ODINDATA_ROOT_DIR}\n")
file(APPEND ${TRISTAN_EXEC_SCRIPT} "./bin/frameProcessor --path ${CMAKE_INSTALL_PREFIX}/lib -d TRISTAN \"$@\"\n")
file(APPEND ${TRISTAN_EXEC_SCRIPT} "cd $cwd\n")

install(
	FILES ${TRISTAN_EXEC_SCRIPT}
	PERMISSIONS OWNER_EXECUTE OWNER_READ
	DESTINATION bin
	)

install(TARGETS TristanProcessPlugin LIBRARY DESTINATION lib)
